# Core views (hosted like plugin views): open the first file of the first Explorer folder in the editor,
# then the terminal. Needs at least one folder open (File > Open Folder…).
launch
wait 2500
eval (async () => { const [f] = await window.oneclient.workspaceFolders(); if (!f) return 'open a folder first'; const kids = await window.oneclient.getExplorerChildren(f.id, null); const file = kids.find((k) => k.nodeType === 'files.file'); if (!file) return 'no file at the top of ' + f.name; await window.oneclient.openWorkspaceFile(file.data.fsPath, { line: 1 }); return 'opened ' + file.label })()
wait 2500
eval-view views/editor (() => document.querySelector('textarea')?.value.length + ' chars')()
press Control+Backquote
wait 2500
eval-view views/terminal [...document.querySelectorAll('.term-item')].map((i) => i.textContent).join(' | ')
quit
